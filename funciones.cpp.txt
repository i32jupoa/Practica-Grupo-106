#include "funciones.h"

// Verifica si el alumno existe en el fichero de inscritos
bool existeAlumno(const std::string &dni, const std::string &fichero) {
    std::ifstream archivo(fichero);
    std::string linea;
    while (getline(archivo, linea)) {
        if (linea.find(dni) != std::string::npos) {
            return true;
        }
    }
    return false;
}

// Verifica si el alumno pertenece a la universidad
bool perteneceAUniversidad(const std::string &dni) {
    std::ifstream archivo("universidad.txt");
    if (!archivo) {
        std::cerr << "Error al abrir el fichero de universidad.\n";
        return false;
    }

    std::string linea;
    while (getline(archivo, linea)) {
        if (linea.find(dni) != std::string::npos) {
            return true;
        }
    }
    return false;
}

// Verifica si el plan existe
bool existePlan(const std::string &nombrePlan) {
    std::ifstream archivo("planes.txt");
    if (!archivo) {
        std::cerr << "Error al abrir el fichero de planes.\n";
        return false;
    }

    std::string linea;
    while (getline(archivo, linea)) {
        if (linea.find(nombrePlan) != std::string::npos) {
            return true;
        }
    }
    return false;
}

// Función para obtener la fecha actual
std::string obtenerFechaActual() {
    std::time_t t = std::time(nullptr);
    char fecha[20];
    std::strftime(fecha, sizeof(fecha), "%Y-%m-%d", std::localtime(&t));
    return std::string(fecha);
}

void inscribirAlumno() {
    Alumno nuevoAlumno;
    std::cout << "Ingrese el nombre: ";
    std::getline(std::cin, nuevoAlumno.nombre);
    std::cout << "Ingrese los apellidos: ";
    std::getline(std::cin, nuevoAlumno.apellidos);
    std::cout << "Ingrese el DNI: ";
    std::getline(std::cin, nuevoAlumno.dni);
    std::cout << "Ingrese el correo electronico: ";
    std::getline(std::cin, nuevoAlumno.correo);

    // Verifica si el alumno pertenece a la universidad
    if (!perteneceAUniversidad(nuevoAlumno.dni)) {
        std::cout << "El alumno debe pertenecer a la universidad para poder inscribirse.\n";
        return;
    }

    // Elige el plan
    std::string plan;
    std::cout << "Seleccione el plan de convalidacion: ";
    std::getline(std::cin, plan);

    // Verifica si el plan existe
    if (!existePlan(plan)) {
        std::cout << "El plan que desea seleccionar no existe.\n";
        return;
    }

    int cuatrimestres;
    std::cout << "¿Desea realizar el plan en 1 o 2 cuatrimestres? (1/2): ";
    std::cin >> cuatrimestres;
    std::cin.ignore();  // Limpiar el buffer

    // Verifica que solo puede ser 1 o 2 cuatrimestres
    if (cuatrimestres != 1 && cuatrimestres != 2) {
        std::cout << "Solo puede seleccionar 1 o 2 cuatrimestres.\n";
        return;
    }

    // Verifica si el alumno ya está inscrito
    if (existeAlumno(nuevoAlumno.dni, "inscritos.txt")) {
        std::cout << "El alumno ya esta inscrito.\n";
        return;
    }

    // Guarda los datos del alumno con el estado "En trámite"
    std::ofstream archivo("inscritos.txt", std::ios::app);
    if (!archivo) {
        std::cerr << "Error al abrir el fichero de inscritos.\n";
        return;
    }

    nuevoAlumno.estadoInscripcion = "En trámite";  // Estado por defecto
    nuevoAlumno.fechaInscripcion = obtenerFechaActual();  // Fecha de inscripción
    archivo << nuevoAlumno.nombre << ", " << nuevoAlumno.apellidos << ", "
            << nuevoAlumno.dni << ", " << nuevoAlumno.correo << ", "
            << nuevoAlumno.estadoInscripcion << ", " << plan << ", "
            << cuatrimestres << " cuatrimestres, Fecha: " << nuevoAlumno.fechaInscripcion << "\n";
    std::cout << "Alumno inscrito correctamente.\n";
}

void agregarPlan() {
    Plan nuevoPlan;
    std::cout << "Ingrese el nombre del plan: ";
    std::getline(std::cin, nuevoPlan.nombre);

    // Verifica si el plan ya existe
    if (existePlan(nuevoPlan.nombre)) {
        std::cout << "El plan de convalidación ya existe.\n";
        return;
    }

    std::cout << "Ingrese la universidad del plan: ";
    std::getline(std::cin, nuevoPlan.universidad);

    std::string asignatura;
    std::cout << "Ingrese las asignaturas (separadas por coma): ";
    std::getline(std::cin, asignatura);

    // Dividimos las asignaturas por coma
    size_t pos = 0;
    while ((pos = asignatura.find(',')) != std::string::npos) {
        nuevoPlan.asignaturas.push_back(asignatura.substr(0, pos));
        asignatura.erase(0, pos + 1);
    }
    nuevoPlan.asignaturas.push_back(asignatura); // Última asignatura

    // Guarda el plan
    std::ofstream archivo("planes.txt", std::ios::app);
    if (!archivo) {
        std::cerr << "Error al abrir el fichero de planes.\n";
        return;
    }

    archivo << nuevoPlan.nombre << ", " << nuevoPlan.universidad << ", ";
    for (const auto &asig : nuevoPlan.asignaturas) {
        archivo << asig << " ";
    }
    archivo << "\n";

    std::cout << "Plan agregado correctamente.\n";
}

void consultarInscripcion() {
    std::string dni;
    std::cout << "Ingrese el DNI del alumno a buscar: ";
    std::cin.ignore();
    std::getline(std::cin, dni);

    std::ifstream archivo("inscritos.txt");
    if (!archivo) {
        std::cerr << "Error al abrir el fichero de inscritos.\n";
        return;
    }

    std::string linea;
    bool encontrado = false;
    while (getline(archivo, linea)) {
        if (linea.find(dni) != std::string::npos) {
            // Solo mostrar el estado de la inscripción
            size_t estadoPos = linea.rfind("En trámite");
            if (estadoPos == std::string::npos) {
                estadoPos = linea.rfind("Aceptada");
            }
            if (estadoPos == std::string::npos) {
                estadoPos = linea.rfind("Rechazada");
            }
            std::cout << "Estado de la inscripción: " << linea.substr(estadoPos) << "\n";
            encontrado = true;
            break;
        }
    }

    if (!encontrado) {
        std::cout << "Alumno no encontrado.\n";
    }
}

void mostrarInscritos() {
    std::ifstream archivo("inscritos.txt");
    if (!archivo) {
        std::cerr << "Error al abrir el fichero de inscritos.\n";
        return;
    }

    std::string linea;
    std::cout << "\nLista de alumnos inscritos:\n";
    while (getline(archivo, linea)) {
        std::cout << linea << "\n";
    }
}

void pruebasErrores() {
    std::cout << "\n--- Pruebas de errores ---\n";

    std::cout << "Prueba 1: Intentar inscribir un alumno con un plan inexistente.\n";
    inscribirAlumno();

    std::cout << "Prueba 2: Agregar un plan con nombre repetido.\n";
    agregarPlan();

    std::cout << "Prueba 3: Consultar un alumno inexistente.\n";
    consultarInscripcion();

    std::cout << "Prueba 4: Mostrar inscritos con archivo vacio.\n";
    std::ofstream limpiar("inscritos.txt");
    limpiar.close();
    mostrarInscritos();
}
